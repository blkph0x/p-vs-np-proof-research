"""Isolated SAT/proof lifecycle probe; never changes the pinned installation."""
import argparse
import json
from pathlib import Path
import sys


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("request", type=Path)
    parser.add_argument("mode", choices=("no_proof", "unread_proof", "read_proof", "flush_proof"))
    parser.add_argument("--pysat", type=Path, required=True)
    args = parser.parse_args()
    sys.path.insert(0, str(args.pysat))
    import pysat
    import pysolvers
    from pysat.solvers import Glucose3
    original = json.loads(args.request.read_text(encoding="utf-8"))
    print(json.dumps({"stage": "start", "version": pysat.__version__,
                      "extension": pysolvers.__file__, "mode": args.mode}), flush=True)
    with Glucose3(bootstrap_with=original["clauses"],
                  with_proof=args.mode != "no_proof") as solver:
        result = solver.solve()
        print(json.dumps({"stage": "solved", "result": result,
                          "stats": solver.accum_stats()}), flush=True)
        if result:
            raw_model = solver.get_model()
            model = {abs(lit): lit > 0 for lit in raw_model}
            original_satisfied = all(any(model.get(abs(lit), False) == (lit > 0)
                                         for lit in clause)
                                     for clause in original["clauses"])
            if not original_satisfied:
                raise RuntimeError("original model check failed")
            print(json.dumps({"stage": "model_checked"}), flush=True)
        if args.mode == "read_proof":
            proof = solver.get_proof()
            print(json.dumps({"stage": "proof_read", "lines": len(proof),
                              "bytes": sum(len(line) + 1 for line in proof)}), flush=True)
        elif args.mode == "flush_proof":
            pysolvers.flushpr(solver.prfp)
            print(json.dumps({"stage": "proof_flushed"}), flush=True)
    print(json.dumps({"stage": "solver_deleted"}), flush=True)
    print(json.dumps({"stage": "normal_main_return"}), flush=True)


if __name__ == "__main__":
    main()
